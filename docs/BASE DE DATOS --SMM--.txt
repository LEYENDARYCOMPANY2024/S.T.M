create table public.terapeutas (
  id_terapeuta serial not null,
  nombre_usuario character varying(100) not null,
  nombre character varying(100) not null,
  primer_apellido character varying(100) not null,
  segundo_apellido character varying(100) null,
  correo character varying(255) not null,
  contrasena text not null,
  documento_cedula bytea null,
  codigo_postal character varying(10) null,
  calle character varying(255) null,
  colonia character varying(255) null,
  num_ext character varying(10) null,
  num_int character varying(10) null,
  foto_perfil bytea null,
  constraint terapeutas_pkey primary key (id_terapeuta),
  constraint terapeutas_correo_key unique (correo),
  constraint terapeutas_nombre_usuario_key unique (nombre_usuario)
) TABLESPACE pg_default;

create table public.terapeuta_paciente (
  id_terapeuta_paciente serial not null,
  id_terapeuta_tp_fk integer not null,
  id_paciente_tp_fk integer not null,
  constraint terapeuta_paciente_pkey primary key (id_terapeuta_paciente),
  constraint terapeuta_paciente_id_paciente_tp_fk_fkey foreign KEY (id_paciente_tp_fk) references pacientes (id_paciente) on delete CASCADE,
  constraint terapeuta_paciente_id_terapeuta_tp_fk_fkey foreign KEY (id_terapeuta_tp_fk) references terapeutas (id_terapeuta) on delete CASCADE
) TABLESPACE pg_default;

create index IF not exists idx_terapeuta_paciente on public.terapeuta_paciente using btree (id_terapeuta_tp_fk, id_paciente_tp_fk) TABLESPACE pg_default;


create table public.sesiones (
  id_sesion serial not null,
  descripcion_sesion text not null,
  fecha_inicio date not null,
  hora_inicio timestamp with time zone not null,

  video_mediapipe_t jsonb not null,
  finalizada boolean null default false,
  constraint sesiones_pkey primary key (id_sesion)
) TABLESPACE pg_default;

create index IF not exists idx_sesiones_fecha on public.sesiones using btree (fecha_inicio) TABLESPACE pg_default;

create table public.sesion_dispositivo (
  id_sesion_dispositivo serial not null,
  id_sesion_fk integer not null,
  id_dispositivo_pd_fk integer not null,
  estado_activado public.estado_dispositivo_enum null default 'desactivado'::estado_dispositivo_enum,
  constraint sesion_dispositivo_pkey primary key (id_sesion_dispositivo),
  constraint sesion_dispositivo_id_dispositivo_pd_fk_fkey foreign KEY (id_dispositivo_pd_fk) references paciente_dispositivo (id_paciente_dispositivo) on delete CASCADE,
  constraint sesion_dispositivo_id_sesion_fk_fkey foreign KEY (id_sesion_fk) references sesiones (id_sesion) on delete CASCADE
) TABLESPACE pg_default;

create unique INDEX IF not exists idx_dispositivo_sesion on public.sesion_dispositivo using btree (id_dispositivo_pd_fk) TABLESPACE pg_default
where
  (
    estado_activado = 'activado'::estado_dispositivo_enum
  );

create table public.pacientes (
  id_paciente serial not null,
  nombre character varying(100) not null,
  primer_apellido character varying(100) not null,
  segundo_apellido character varying(100) null,
  edad integer null,
  codigo character varying(50) not null,
  genero public.genero_enum not null,
  estado_civil public.estado_civil_enum null,
  escolaridad public.escolaridad_enum null,
  ocupacion public.ocupacion_enum null,
  nombre_usuario_p character varying(100) null,
  contrasena_p text null,
  estado_paciente public.estado_paciente_enum not null,
  correo_electronico_p text null,
  foto_perfil bytea null,
  codigo_restablecimiento character varying null,
  fecha_expiracion_codigo timestamp with time zone null,
  constraint pacientes_pkey primary key (id_paciente),
  constraint pacientes_codigo_key unique (codigo),
  constraint pacientes_nombre_usuario_p_key unique (nombre_usuario_p),
  constraint pacientes_edad_check check (
    (
      (edad > 0)
      and (edad < 120)
    )
  )

create index IF not exists idx_pacientes_nombre on public.pacientes using btree (nombre, primer_apellido) TABLESPACE pg_default;
